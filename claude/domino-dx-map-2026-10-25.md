## ה. מלאי אירועים — SQL
```sql
-- 2026-10-25: Momentum / Pixie / Puzzle events, 2026-04-02..2026-10-12 (domino-dreams-dbt, 43 GB)
WITH m AS (
  SELECT 'momentum' AS feat, momentum_event_id AS ev_id, ANY_VALUE(mtx_event_name) AS ev_name, ANY_VALUE(mtx_campaign_name) AS camp,
    MIN(momentum_event_start_date) AS s, MAX(momentum_event_end_date) AS e, COUNT(DISTINCT user_id) AS users,
    COUNT(DISTINCT IF(is_completed, user_id, NULL)) AS done, MAX(max_tiers) AS tiers
  FROM `domino-dreams-dbt.prod_40_mart.user_daily_momentum`
  WHERE date_dim BETWEEN DATE '2026-04-02' AND DATE '2026-10-12' GROUP BY 2
),
p AS (
  SELECT 'pixie', pixie_event_id, CAST(ANY_VALUE(album_id) AS STRING), '', MIN(pixie_event_start_date), MAX(pixie_event_end_date),
    COUNT(DISTINCT user_id), NULL, MAX(pixie_stickers_lt)
  FROM `domino-dreams-dbt.prod_40_mart.user_daily_pixie`
  WHERE date_dim BETWEEN DATE '2026-04-02' AND DATE '2026-10-12' GROUP BY 2
),
z AS (
  SELECT 'puzzle', puzzle_event_id, ANY_VALUE(mtx_event_name), ANY_VALUE(mtx_campaign_name), MIN(puzzle_event_start_date), MAX(puzzle_event_end_date),
    COUNT(DISTINCT user_id), COUNT(DISTINCT IF(puzzle_completed_flag, user_id, NULL)), MAX(max_tier_reached)
  FROM `domino-dreams-dbt.prod_40_mart.user_daily_puzzle`
  WHERE date_dim BETWEEN DATE '2026-04-02' AND DATE '2026-10-12' GROUP BY 2
)
SELECT feat, ev_id, ev_name, camp, s, e, users, done, tiers FROM m
UNION ALL SELECT * FROM p UNION ALL SELECT * FROM z ORDER BY feat, s;
```
